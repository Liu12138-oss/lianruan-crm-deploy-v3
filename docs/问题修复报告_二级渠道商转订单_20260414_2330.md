**修复时间**: 2026-04-14 23:30

## 问题描述

### 用户反馈
二级渠道商报价单转订单时，提示错误：
> "创建订单失败：二级渠道商必须向绑定的一级渠道商下单"

### 用户期望
二级渠道商报价转订单时，应该：
1. 弹出选择一级渠道商的对话框
2. 显示其绑定的一级渠道商信息
3. 让用户选择或确认
4. 允许提交订单，而不是直接拒绝

---

## 问题分析

### 根本原因

后端API的设计是：
1. 二级渠道商必须通过一级渠道商下单
2. 调用API时需要传递 `parentPartnerId`（一级渠道商ID）
3. 如果不传递或传递错误，后端会拒绝请求

前端代码的问题：
1. 调用 `apiClient.createOrder` 时没有传递 `parentPartnerId`
2. 直接使用当前渠道商的ID作为 `partnerId`
3. 导致后端校验失败

### 错误代码追踪

**前端调用**（partner-app.js）：
```javascript
// 报价单转订单
const result = await apiClient.createOrder({
  quoteId: q.id,
  customer: q.customer,
  total: q.total,
  partnerId: store.user?.partnerId || '',  // ❌ 使用了当前渠道商ID
  // ❌ 缺少 parentPartnerId 参数
});
```

**后端校验**（server.js）：
```javascript
app.post('/api/orders', (req, res) => {
  const currentPartnerId = req.body.partnerId;
  const currentPartner = db.partners.find(p => p.id === currentPartnerId);
  
  if (currentPartner?.partnerLevel === 'secondary') {
    // 二级渠道商下单时，必须指定 parentPartnerId
    if (!req.body.parentPartnerId) {
      return res.status(400).json({
        success: false,
        error: '二级渠道商必须向绑定的一级渠道商下单'
      });
    }
    
    // 校验 parentPartnerId 是否是一级渠道商
    const parent = db.partners.find(p => p.id === req.body.parentPartnerId);
    if (parent?.partnerLevel !== 'primary') {
      return res.status(400).json({
        success: false,
        error: '指定的一级渠道商无效'
      });
    }
  }
});
```

```javascript
// 打开选择一级渠道商弹窗
function openSelectParentPartnerModal(q) {
  const currentPartner = store.partners?.find(p => p.id === store.user.partnerId);
  if (!currentPartner || currentPartner.partnerLevel !== 'secondary') {
    // 不是二级渠道商，直接创建订单
    toOrderDirectly(q);
    return;
  }
  
  // 获取绑定的一级渠道商
  const parentPartnerId = currentPartner.parentPartnerId;
  if (!parentPartnerId) {
    alert('您未绑定一级渠道商，请联系管理员设置');
    return;
  }
  
  const parentPartner = store.partners?.find(p => p.id === parentPartnerId);
  if (!parentPartner) {
    alert('绑定的一级渠道商信息不存在，请联系管理员');
    return;
  }
  
  selectedParentPartner.value = parentPartner;
  currentQuoteForParentSelect.value = q;
  showParentPartnerModal.value = true;
}
```

#### 3. 修改报价单转订单方法
```javascript
async function toOrder(q) {
  const existing = store.orders.find(o => o.quoteId === q.id);
  if (existing) { alert('该报价单已转为订单：' + existing.id); return; }
  
  // 检查是否为二级渠道商
  const currentPartner = store.partners?.find(p => p.id === store.user.partnerId);
  if (currentPartner?.partnerLevel === 'secondary') {
    // 二级渠道商：打开选择一级渠道商弹窗
    openSelectParentPartnerModal(q);
  } else {
    // 非二级渠道商：直接创建订单
    await toOrderDirectly(q);
  }
}
```

#### 4. 添加确认选择并创建订单的方法
```javascript
async function confirmParentAndCreateOrder() {
  if (!selectedParentPartner.value) {
    alert('请选择一级渠道商');
    return;
  }
  
  const q = currentQuoteForParentSelect.value;
  const reg = store.registrations.find(r => r.id === q.regId);
  showParentPartnerModal.value = false;
  
  try {
    const result = await apiClient.createOrder({
      quoteId: q.id,
      customer: q.customer,
      total: q.total,
      deliveryAddr: deliveryForm.addr?.trim() || (reg ? reg.city || '' : ''),
      contacts: reg ? `${reg.contact} ${reg.phone}` : '待填写',
      region: reg ? reg.region : (q.region || ''),
      regId: q.regId || '',
      oppId: q.oppId || '',
      partnerId: store.user?.partnerId || '',
      partnerName: store.user?.partnerName || store.user?.name || '',
      parentPartnerId: selectedParentPartner.value.id,      // ✅ 传递选择的一级渠道商ID
      parentPartnerName: selectedParentPartner.value.name,  // ✅ 传递一级渠道商名称
      createdBy: q.createdBy || store.user.id,
      assignedStaffId: q.assignedStaffId || store.user.id,
    });
    
    if (result.success) {
      store.orders.unshift(result.data);
      alert('已成功转为订单，请在订单管理中查看。');
      router.push('/order');
    } else {
      alert('创建订单失败：' + (result.error || '未知错误'));
    }
  } catch (err) {
    alert('创建订单失败，请检查网络连接');
  }
}
```
